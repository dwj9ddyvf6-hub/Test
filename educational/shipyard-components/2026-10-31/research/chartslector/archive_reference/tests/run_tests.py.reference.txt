def test(system, name):
    def deco(f):
        def run():
            t0 = time.time()
            try: msg = f() or ""; ok = True
            except AssertionError as e: ok = False; msg = "ASSERT: " + str(e)
            except Exception as e: ok = False; msg = "ERROR: " + repr(e) + " | " + traceback.format_exc().splitlines()[-3]
            RES.append(dict(system=system, test=name, ok=ok, secs=round(time.time() - t0, 1), detail=str(msg)[:400])); print(("PASS " if ok else "FAIL ") + system + " :: " + name + " :: " + str(msg)[:300], flush=True)
        run.__name__ = name; run.system = system; TESTS.append(run); return run
    return deco

def close(a, b, tol, what):
    a = np.asarray(a, float); b = np.asarray(b, float); m = np.isfinite(a) & np.isfinite(b)
    assert m.sum() > 0, f"{what}: no overlapping finite values"
    d = np.abs(a[m] - b[m]).max(); assert d <= tol, f"{what}: max abs diff {d:.3g} > {tol}"
    return f"{what}: {m.sum()} bars, max diff {d:.2g}"
