RES = []
def test(name):
    def deco(f):
        if os.environ.get("CL_TEST_ONLY") and os.environ["CL_TEST_ONLY"] not in name: return f
        t0 = time.time()
        try: msg = f() or ""; ok = True
        except AssertionError as e: ok, msg = False, "ASSERT " + str(e)
        except Exception as e: ok, msg = False, "ERROR " + repr(e) + " " + traceback.format_exc().splitlines()[-2]
        RES.append(dict(test=name, ok=ok, detail=str(msg)[:300], secs=round(time.time() - t0, 1))); print("PASS" if ok else "FAIL", name, "::", str(msg)[:250], flush=True); return f
    return deco
def sha(p): return hashlib.sha256(open(p, "rb").read()).hexdigest()

@test("ship + build + rollback rehearsal on a copy of the STAGED tree (byte-identical restore)")
def _():
    if os.path.exists(RH): shutil.move(RH, RH + f"_old_{int(time.time())}")
    # mirror the real layout <root>/shadow/paper_sleeve_STAGED_1.0.4: the builder reads {shadow}/plans_calibrated_v1_frozen.json and {root}/CORPUS_AUDIT, {root}/DESK_INTEGRATION (read-only -> symlinks)
    os.makedirs(f"{RH}/shadow"); T = f"{RH}/shadow/paper_sleeve_STAGED_1.0.4"; shutil.copytree(STAGED, T, symlinks=True)
    os.symlink("__STAGED_PLAN_FILE__", f"{RH}/shadow/plans_calibrated_v1_frozen.json")
    for d in ("CORPUS_AUDIT", "DESK_INTEGRATION"): os.symlink(f"__OWNER_ARCHIVE_PARENT__/{d}", f"{RH}/{d}")
    pre = ""
    if os.path.exists(f"{T}/chartslector"):   # STAGED already has the CL package (installed 2026-10-06): roll the copy back first, then rehearse ship -> build -> rollback from that base
        rb0 = subprocess.run(["bash", f"{PKG}/rollback_sleeve_cl.sh"], env=dict(os.environ, SLEEVE_STAGED=T, CL_NO_GLOBAL_LOG="1"), capture_output=True, text=True); assert rb0.returncode == 0, rb0.stdout[-300:] + rb0.stderr[-300:]
        pre = "installed STAGED copy rolled back first (rollback OK); "
    before = {rel: sha(f"{T}/{rel}") for rel in ("build/lane_budgets.json", "build/build_sleeve_playbook.py", "sleeve_scheduler.sh", "sleeve_playbook.json")}
    env = dict(os.environ, SLEEVE_STAGED=T, ACTIVATE="1", CL_NO_GLOBAL_LOG="1")
    r = subprocess.run(["bash", f"{PKG}/ship_sleeve_cl.sh"], env=env, capture_output=True, text=True); assert r.returncode == 0, r.stdout[-300:] + r.stderr[-300:]
    r2 = subprocess.run(["bash", f"{PKG}/ship_sleeve_cl.sh"], env=env, capture_output=True, text=True); assert r2.returncode == 2, "second ship must refuse"
    os.chmod(f"{T}/sleeve_playbook.json", 0o644)  # the staged playbook is frozen (444); a real rebuild goes through the staged tree's own build + freeze with a new version string
    b = subprocess.run([PY, f"{T}/build/build_sleeve_playbook.py"], capture_output=True, text=True, cwd=T); assert b.returncode == 0, b.stderr.strip().splitlines()[-1][-300:]
    PB = json.load(open(f"{T}/sleeve_playbook.json")); Lq, Lv = PB["lanes"]["s_cl_qm_small"], PB["lanes"]["s_cl_vmr_qqq"]
    o = json.load(open(f"{PKG}/build/option_v3b_cl_20261006.json"))["budgets_usd"]
    assert Lq["enabled"] and Lq["entry"]["policy"] == "next_open" and Lq["sizing"]["position_usd"] == 1000 and Lq["sizing"]["budget_usd"] == o["s_cl_qm_small"] and Lq["sizing"]["lane_max_slots"] == o["s_cl_qm_small"] // 1000
    assert Lv["sizing"]["budget_usd"] == o["s_cl_vmr_qqq"]; sch = open(f"{T}/sleeve_scheduler.sh").read(); assert "cl_live_fires" in sch and "return $f" in sch
    shutil.copy(f"{T}/sleeve_playbook.json", f"{P}/rehearsal_sleeve_playbook_with_cl.json")
    rb = subprocess.run(["bash", f"{PKG}/rollback_sleeve_cl.sh"], env=env, capture_output=True, text=True); assert rb.returncode == 0, rb.stderr[-300:]
    # the builder run rewrote sleeve_playbook.json in the copy; restore check covers the 3 shipped files; rebuilding after rollback must give the original playbook bytes back
    b2 = subprocess.run([PY, f"{T}/build/build_sleeve_playbook.py"], capture_output=True, text=True, cwd=T); assert b2.returncode == 0
    after = {rel: sha(f"{T}/{rel}") for rel in before}; diff = [k for k in before if before[k] != after[k]]

    PB2 = json.load(open(f"{T}/sleeve_playbook.json")); assert "s_cl_qm_small" not in PB2["lanes"] and not os.path.exists(f"{T}/chartslector")
